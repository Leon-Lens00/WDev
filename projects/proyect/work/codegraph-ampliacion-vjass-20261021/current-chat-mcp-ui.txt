2	// Coordinates remain UserInterface coordinates; this is a preview transform only.
3	export function projectionRect(view){
4	 if(view.projection!=='classic-127b')return {x:0,y:0,w:view.width,h:view.height};
5	 const w=Math.min(view.width,view.height*4/3);
6	 return {x:(view.width-w)/2,y:view.height*.04,w,h:view.height*.72};
7	}
8	export function previewGeometry(e,view,{textBaseline=true}={}){
9	 const r=projectionRect(view);
10	 const textOffset=textBaseline&&e.type==='text'?(e.font*.0023+.0065)*view.height:0;
11	 return {x:r.x+(e.x+1)*r.w/2,y:r.y+(1-e.y)*r.h/2-textOffset,w:e.w*r.w/2,h:e.h*r.h/2};
12	}
13	export function previewValue(key,value,view){
14	 const r=projectionRect(view);
15	 return key==='x'?(value-r.x)*2/r.w-1:key==='y'?1-(value-r.y)*2/r.h:value*2/(key==='w'?r.w:r.h);
16	}
17	export function previewDelta(dx,dy,view){const r=projectionRect(view);return {x:dx*2/r.w,y:-dy*2/r.h};}
